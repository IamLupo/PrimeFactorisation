  C_r = [x^(r+1)] F.

  C_r = binom(l,r+1)
        * (p^k q^(l-r-1) + q^k p^(l-r-1))
        - 1_(r+1<=k) binom(k,r+1)
        * (p^l q^(k-r-1) + q^l p^(k-r-1))
